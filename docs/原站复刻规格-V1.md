### 2.1 算法（chunk `9993-7a1783218bf3e4c7.js` 模块 39302，已逐宽度复算验证 expMatch=True）

```js
const n = parseFloat(getComputedStyle(document.documentElement).fontSize);   // 16
const a = getComputedStyle(grid);
const l = getComputedStyle(grid.parentElement);      // section
const s = parseFloat(l.paddingLeft) + parseFloat(l.paddingRight);
const o = Math.max(1, (mainShellW - 56 - s) - 2 * parseFloat(a.columnGap));   // 可用宽度
const e = window.innerWidth;
const c = e >= 2200, u = e >= 1800;
const d = (c ? 32 : u ? 30 : 29) * n;                // 编辑列最小 fr 基数
const p = c ? 0.57 : u ? 0.54 : 0.51;               // 编辑列占比

// expanded
const av = c ? 12.5*n
           : u ? Math.min(12.75*n, Math.max(10.5*n, 0.105*e))
           :    Math.min(12.25*n, Math.max(10*n, 0.11*e));          // 右列 fr 基数
const ew = Math.max(d, (o - av) * p / (1 + p));                     // 编辑列 fr
const pw = Math.max(1, o - av - ew);                               // 预览列 fr
// collapsed
const av2 = (u ? 4.75 : 4.5) * n;
const ew2 = Math.max(d, (o - av2) * p / (1 + p));
const pw2 = Math.max(1, o - av2 - ew2);

grid.style.setProperty('--toolbox-expanded-columns',
  `minmax(386px,${ew.toFixed(3)}fr) minmax(0,${pw.toFixed(3)}fr) minmax(150px,${av.toFixed(3)}fr)`);
grid.style.setProperty('--toolbox-collapsed-columns',
  `minmax(386px,${ew2.toFixed(3)}fr) minmax(0,${pw2.toFixed(3)}fr) minmax(0,${av2.toFixed(3)}fr)`);
```
